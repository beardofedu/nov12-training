# INERT REFERENCE ONLY: default setup owns live CodeQL in nov12-training.
# Do not copy into .github/workflows/ or dispatch while default setup is enabled.
# Default setup disables existing CodeQL workflows and rejects advanced uploads:
# https://docs.github.com/en/code-security/reference/code-scanning/troubleshoot-analysis-errors/two-codeql-workflows
# Switching setup modes requires a separate owner decision; not part of this demo.
name: CodeQL

on:
  workflow_dispatch:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  schedule:
    - cron: '30 6 * * 1'

permissions:
  security-events: write
  contents: read
  actions: read

jobs:
  analyze:
    name: Analyze (${{ matrix.language }})
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        language: [javascript-typescript, python, actions]
    steps:
      - uses: actions/checkout@v4
      - uses: github/codeql-action/init@v3
        with:
          languages: ${{ matrix.language }}
          # This query suite is NOT the separate GitHub Code Quality product.
          queries: security-and-quality
      - uses: github/codeql-action/analyze@v3
        with:
          category: /language:${{ matrix.language }}
